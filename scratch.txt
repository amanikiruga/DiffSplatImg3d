class CleanGaussianVoxelGrid(th.nn.Module):
    """
    Clean voxel grid class for rendering using the new normalization approach.
    """
    def __init__(self, grid_size=32, max_sh_degree=3, device="cuda"):
        super().__init__()
        self.grid_size = grid_size
        self.max_sh_degree = max_sh_degree
        
        # Create fixed voxel centers
        half_extent = 0.5
        coords = th.linspace(-half_extent, half_extent, grid_size, device=device)
        x, y, z = th.meshgrid(coords, coords, coords, indexing="ij")
        voxel_centers = th.stack([x, y, z], dim=-1).reshape(grid_size, grid_size, grid_size, 3)
        self.register_buffer("voxel_centers", voxel_centers)

    def load_from_volume(self, volume, include_features, norm_stats):
        """
        Load parameters from a normalized volume tensor using clean denormalization.
        
        Args:
            volume: [C, D, H, W] normalized tensor in [-1,1] range
            include_features: List of feature names
            norm_stats: Global normalization statistics
        """
        # Denormalize the volume
        denorm_volume = denormalize_gaussian_volume(volume.detach(), include_features, norm_stats)
        
        # Convert to gaussian dict
        gauss_dict = volume_to_gaussian_dict(
            denorm_volume, include_features, self.grid_size, 
            self.voxel_centers, opacity_threshold=0.01
        )
        
        return gauss_dict

    def get_gaussian_parameters_at_voxels(self, volume, include_features, norm_stats, opacity_threshold=0.01):
        """Convert normalized volume to gaussian parameters for rendering."""
        print(f"=== GET_GAUSSIAN_PARAMETERS DEBUG ===")
        print(f"Input volume shape: {volume.shape}")
        print(f"Input volume range: [{volume.min().item():.3f}, {volume.max().item():.3f}]")
        print(f"Include features: {include_features}")
        print(f"Opacity threshold: {opacity_threshold}")
        
        result = self.load_from_volume(volume, include_features, norm_stats)
        
        print(f"Gaussian parameters created:")
        for key, value in result.items():
            if isinstance(value, th.Tensor):
                print(f"  {key}: shape={value.shape}, range=[{value.min().item():.3f}, {value.max().item():.3f}]")
        
        return result

def render_voxel_views(volume, include_features, norm_stats, dataset_obj, cfg, device, render_config, num_views=8, opacity_threshold=0.01, is_noise=False):
    """
    Render multiple views of a normalized voxel volume.
    
    Args:
        volume: [C, D, H, W] normalized volume tensor
        include_features: List of feature names
        norm_stats: Global normalization statistics
        dataset_obj: Dataset object with camera information
        cfg: Rendering configuration
        device: torch device
        num_views: Number of views to render
        opacity_threshold: Opacity threshold for rendering
    
    Returns:
        List of rendered images as numpy arrays
    """
    print(f"=== RENDER_VOXEL_VIEWS DEBUG ===")
    print(f"RENDERING_AVAILABLE: {RENDERING_AVAILABLE}")
    print(f"Volume shape: {volume.shape}, range: [{volume.min().item():.3f}, {volume.max().item():.3f}]")
    print(f"Include features: {include_features}")
    print(f"Opacity threshold: {opacity_threshold}")
    print(f"Dataset object keys: {list(dataset_obj.keys()) if dataset_obj else 'None'}")
    
    if not RENDERING_AVAILABLE:
        print("Warning: Rendering not available, returning dummy images")
        return [np.zeros((128, 128, 3), dtype=np.uint8) for _ in range(num_views)]
    
    bg_color = [1, 1, 1] if cfg.data.white_background else [0, 0, 0]
    background = th.tensor(bg_color, dtype=th.float32, device=device)
    print(f"Background color: {bg_color}")
    
    # Create voxel grid and get gaussian parameters
    voxel_grid = CleanGaussianVoxelGrid(device=device)
    if not is_noise:
        print("pupu go huhu")
        gauss = voxel_grid.get_gaussian_parameters_at_voxels(volume, include_features, norm_stats, opacity_threshold)
    else:
        print("papa go haha")
        gauss = volume_to_gaussian_dict(volume, include_features, voxel_grid.grid_size, voxel_grid.voxel_centers, opacity_threshold=opacity_threshold)
    
    print(f"Gaussian parameters after denormalization:")
    for key, value in gauss.items():
        if isinstance(value, th.Tensor):
            print(f"  {key}: shape={value.shape}, range=[{value.min().item():.3f}, {value.max().item():.3f}], mean={value.mean().item():.3f}")

    # Select views to render
    total_views = dataset_obj['gt_images'].shape[0]
    
    view_indices = th.linspace(0, total_views-1, num_views).long()
    print(f"Total views available: {total_views}, selected indices: {view_indices.tolist()}")
    
    frames = []
    with th.no_grad():
        for i, v in enumerate(view_indices):
            try:
                if RENDERING_AVAILABLE and dataset_obj and 'world_view_transforms_absolute' in dataset_obj:
                    print(f"Rendering view {i}/{len(view_indices)} (index {v})...")
                    out = render_predicted(
                        gauss,
                        dataset_obj['world_view_transforms_absolute'][v],
                        dataset_obj['full_proj_transforms_absolute'][v], 
                        dataset_obj['camera_centers_absolute'][v],
                        background, render_config
                    )
                    img = out['render'].cpu().numpy().transpose(1, 2, 0)
                    print(f"Rendered image shape: {img.shape}, range: [{img.min():.3f}, {img.max():.3f}]")
                    
                    # Check if image is all black
                    if img.max() == 0:
                        print(f"WARNING: Rendered image {i} is completely black!")
                    
                    frames.append((np.clip(img, 0, 1) * 255).astype(np.uint8))
                else:
                    print(f"Fallback: Creating dummy image for view {i} (missing camera data)")
                    frames.append(np.zeros((128, 128, 3), dtype=np.uint8))
            except Exception as e:
                print(f"ERROR: Rendering view {v} failed: {e}")
                raise e
                import traceback
                traceback.print_exc()
    
    print(f"=== RENDER COMPLETE: {len(frames)} frames ===")
    return frames


def create_multiview_grid(frames):
    """Create a grid of multiview images with dynamic size."""
    if not frames:
        raise ValueError("No frames to create grid from")
    
    num_frames = len(frames)
    grid_size = math.ceil(math.sqrt(num_frames))
    
    # Pad frames to fill grid
    while len(frames) < grid_size * grid_size:
        frames.append(np.zeros_like(frames[0]))
    
    # Arrange in grid
    rows = []
    for i in range(grid_size):
        row_frames = frames[i*grid_size:(i+1)*grid_size]
        rows.append(np.concatenate(row_frames, axis=1))
    
    return np.concatenate(rows, axis=0)

    def forward_backward(self, batch, cond):
        self.mp_trainer.zero_grad()
        for i in range(0, batch.shape[0], self.microbatch):
            micro = batch[i : i + self.microbatch].to(dist_util.dev())
            micro_cond = {
                k: v[i : i + self.microbatch].to(dist_util.dev())
                for k, v in cond.items()
            }
            last_batch = (i + self.microbatch) >= batch.shape[0]
            t, weights = self.schedule_sampler.sample(micro.shape[0], dist_util.dev())
            
            # Custom loss computation with alpha_bar^2 weighting
            def compute_losses():
                noise = th.randn_like(micro)
                x_t = self.diffusion.q_sample(micro, t, noise=noise)
                
                # Log training visualizations only at log intervals
                if self.step % self.log_interval == 0:
                    # Log ground truth (clean training data) 
                    self.log_voxel_viz(micro[0], f"train/gt_data")
                    # Log noisy input 
                    self.log_voxel_viz(x_t[0], f"train/noisy_input")
                
                terms = {}
                model_output = self.ddp_model(x_t, t, **micro_cond)
                
                # Log denoised output only at log intervals
                if self.step % self.log_interval == 0:
                    self.log_voxel_viz(model_output[0], f"train/model_output")
                
                target = {
                    gd.ModelMeanType.PREVIOUS_X: self.diffusion.q_posterior_mean_variance(
                        x_start=micro, x_t=x_t, t=t
                    )[0],
                    gd.ModelMeanType.START_X: micro,
                    gd.ModelMeanType.EPSILON: noise,
                }[self.diffusion.model_mean_type]
                
                assert model_output.shape == target.shape == micro.shape
                
                # Compute MSE loss
                mse_loss = ((target - model_output) ** 2).mean(dim=list(range(1, len(target.shape))))
                
                # Apply custom weighting: ωt = ᾱ²t
                alpha_bar_t = self.alpha_bar_tensor[t]
                custom_weights = alpha_bar_t ** 2
                
                terms["mse"] = mse_loss
                terms["loss"] = mse_loss * custom_weights
                return terms

            if last_batch or not self.use_ddp:
                losses = compute_losses()
            else:
                with self.ddp_model.no_sync():
                    losses = compute_losses()

            loss = (losses["loss"] * weights).mean()
            
            # Log to wandb only on primary rank
            if self.is_primary_rank:
                wandb.log({
                    "train/loss": loss.item(),
                    "train/mse": (losses["mse"] * weights).mean().item(),
                    "train/step": self.step,
                    "train/samples": self.step * self.batch_size * self.world_size,  # Account for all ranks
                })
            
            # Log losses to logger as well (all ranks for aggregation)
            logger.logkv_mean("train_loss", loss.item())
            logger.logkv_mean("train_mse", (losses["mse"] * weights).mean().item())
            
            self.mp_trainer.backward(loss)

def main():
    args = create_argparser().parse_args()

    # Setup distributed training first
    dist_util.setup_dist()
    logger.configure()
    
    # Get distributed training info
    try:
        from mpi4py import MPI
        rank = MPI.COMM_WORLD.Get_rank()
        world_size = MPI.COMM_WORLD.Get_size()
    except ImportError:
        rank = 0
        world_size = 1
    
    logger.log(f"Distributed training: rank {rank}/{world_size}")

    # Setup wandb only on rank 0 to avoid conflicts
    if rank == 0:
        wandb.init(
            project=args.wandb_project,
            name=args.wandb_run,
            config=args_to_dict(args, voxel_gaussian_model_and_diffusion_defaults().keys()),
            tags=["voxel-gaussian", "3d-diffusion", "clean-normalization", f"world_size_{world_size}"]
        )
        
        # Log all training parameters
        wandb.config.update({
            "gaussian_features": args.include_features,
            "expected_channels": calculate_voxel_gaussian_channels(args.include_features),
            "grid_size": args.volume_size,
            "custom_loss_weighting": "alpha_bar_squared",

        norm_stats_path=args.norm_stats_path,
    )
    
    # Run training with periodic test generation
    original_run_loop = train_loop.run_loop
    
    def enhanced_run_loop():
        # Log test generation at start
        if args.log_test_generation:
            train_loop.log_test_generation()
        
        # Override run_step to add periodic test generation and max_steps
        original_run_step = train_loop.run_step
        
        def enhanced_run_step(batch, cond):
            # Check if we've reached max_steps
            if args.max_steps is not None and train_loop.step >= args.max_steps:
                print(f"Reached max_steps ({args.max_steps}), stopping training")
                return False
            
            result = original_run_step(batch, cond)
            
            # Log test generation periodically at specified intervals
            if (args.log_test_generation and 
                train_loop.step > 0 and 
                train_loop.step % args.test_generation_interval == 0):
                print(f"Generating test samples at step {train_loop.step}")
                train_loop.log_test_generation()
            
            return result
        
        train_loop.run_step = enhanced_run_step
        
        # Override run_loop to respect max_steps
        if args.max_steps is not None:
            while train_loop.step < args.max_steps:
                try:
                    batch, cond = next(train_loop.data)
                    if enhanced_run_step(batch, cond) == False:
                        break
                except StopIteration:
                    print("Data iterator exhausted")
                    break
        else:
            return original_run_loop()
    
    train_loop.run_loop = enhanced_run_loop
    train_loop.run_loop()

import torch, numpy as np
from pathlib import Path

CKPT_ROOT = "..."   # your root
N_CH = 4
DELTA = 1.0        # set to your voxel step length
EPS = 1e-6
PCT = 99.5         # robust high-percentile for scaling

def _sample_abs_percentile(values, q, max_samples=2_000_000, rng=np.random.default_rng(0)):
    # values: 1D numpy array; returns qth percentile of |values|
    n = values.size
    if n <= max_samples:
        v = np.abs(values.ravel())
    else:
        idx = rng.integers(0, n, size=max_samples, endpoint=False)
        v = np.abs(values.ravel()[idx])
    s = np.percentile(v, q)
    return max(float(s), EPS)

def compute_norm_params(ckpt_root: str):
    paths = sorted(Path(ckpt_root).rglob("dense_grid.npz"))
    if not paths:
        raise FileNotFoundError("No dense_grid.npz found.")

    # For density: estimate scale in LOGIT-OPACITY space
    dens_logits = []
    # For SH: per-channel robust scales (linear+clip)
    sh_scales_accum = [[] for _ in range(3)]  # channels 1,2,3

    rng = np.random.default_rng(0)
    for p in paths:
        g = np.load(p)["dense_grid"]  # (X,Y,Z,4)
        d = g[..., 0].astype(np.float64)
        sigma = np.maximum(d, 0.0)                      # or: softplus(d)
        alpha = 1.0 - np.exp(-sigma * DELTA)
        ell = np.log(alpha + EPS) - np.log(1.0 - alpha + EPS)
        dens_logits.append(ell)

        for i in range(3):
            sh_scales_accum[i].append(g[..., 1 + i].astype(np.float64))

    # density scale = percentile of |logit(alpha)|
    dens_logits_flat = np.concatenate([x.ravel() for x in dens_logits])
    dens_scale = _sample_abs_percentile(dens_logits_flat, PCT, rng=rng)

    # SH scales = per-channel abs-percentile
    sh_scales = []
    for i in range(3):
        flat = np.concatenate([x.ravel() for x in sh_scales_accum[i]])
        sh_scales.append(_sample_abs_percentile(flat, PCT, rng=rng))

    params = {
        "density": {"scale": float(dens_scale), "delta": float(DELTA), "eps": float(EPS)},
        "sh_scale": torch.tensor(sh_scales, dtype=torch.float32),  # shape [3]
    }
    torch.save(params, "norm_params.pt")
    return params

def normalize(grid: torch.Tensor, params: dict) -> torch.Tensor:
    g = grid.float()
    # density → opacity logit → scaled & clipped to [-1,1]
    d = g[..., 0]
    sigma = torch.relu(d)                                 # or: F.softplus(d)
    alpha = 1.0 - torch.exp(-sigma * params["density"]["delta"])
    ell = torch.log(alpha + params["density"]["eps"]) - torch.log(1 - alpha + params["density"]["eps"])
    dens_norm = torch.clamp(ell / params["density"]["scale"], -1.0, 1.0)

    # SH: linear scale by robust percentile and clip
    sh = g[..., 1:4]
    sh_scale = params["sh_scale"].to(sh.device).view(*([1] * (sh.dim() - 1)), 3)
    sh_norm = torch.clamp(sh / sh_scale, -1.0, 1.0)

    return torch.cat([dens_norm.unsqueeze(-1), sh_norm], dim=-1)

def denormalize(norm_grid: torch.Tensor, params: dict) -> torch.Tensor:
    ng = norm_grid.float().clamp(-1, 1)
    # density: invert scale → sigmoid → invert opacity to sigma
    ell = ng[..., 0] * params["density"]["scale"]
    alpha = torch.sigmoid(ell)
    sigma = -torch.log(1.0 - alpha + params["density"]["eps"]) / params["density"]["delta"]
    d_rec = sigma  # keep non-negative; if you need pre-activation, you can store sigma directly

    # SH: invert linear scale
    sh = ng[..., 1:4]
    sh_scale = params["sh_scale"].to(sh.device).view(*([1] * (sh.dim() - 1)), 3)
    sh_rec = sh * sh_scale

    return torch.cat([d_rec.unsqueeze(-1), sh_rec], dim=-1)
